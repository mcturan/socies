📐 Titreşim Sensörü (SW-420) & Güncel Donanım Şeması

Cihaz kutusuna eklenen SW-420 Titreşim Sensörü (veya basit bilyalı tilt anahtarı / ball rolling switch), kullanıcı cihazı cebinde salladığında veya yürürken adımlarını algılar. Harici kesme (Interrupt) pini üzerinden ESP32 uyanır ve adım sayar.

💻 ESP32 C++ Kodu (Titreşim Sensörü Kesmesi & Gece Modu)

// SW-420 Titreşim Sensörü ve Adım Sayar Kesme Kodu
#define PIN_VIBRO_SENSOR 7
#define PIN_VIBRO_MOTOR  6
#define PIN_BUZZER       4

volatile unsigned long lastShakeTime = 0;
volatile uint32_t stepCounter = 0;

void IRAM_ATTR onShakeInterrupt() {
  unsigned long now = millis();
  if (now - lastShakeTime > 180) { // 180ms debounce filtresi
    stepCounter++;
    lastShakeTime = now;
    // Oyun ihtiyacı artar
    sosis.fun = min(100, sosis.fun + 2);
  }
}

void setup() {
  pinMode(PIN_VIBRO_SENSOR, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(PIN_VIBRO_SENSOR), onShakeInterrupt, FALLING);
}
